{
  "name": "performance-optimizer",

  "publisher": "yourusername",
  "description": "Optimizes your VS Code environment for better performance.",
  "author": {
    "name": "Your Name",
    "url": "https://www.yourwebsite.com"
  },
  "main": "./src/extension.js",
  "scripts": {
    "start": "code --develop",
    "pack": "tsc && npx rollup -c"
  },
  "dependencies": {
    "vscode": "^1.69.0",

    "typescript": "^4.8.3"

  "devDependencies": {

### main code (extension.js)

```javascript
import * as vscode from 'vscode';

export function activate(context: vscode.ExtensionContext) {
  const disposable = vscode.commands.registerCommand('performance-optimizer.disableExtensions', () => {
    vscode.extensions.get()!.forEach(ext => {
      if (!ext.isActive) {
        ext.activate().then(() => {
          console.log(`Extension ${ext.packageJSON.name} is now active.`);
        }).catch(err => {
          console.error(`Error activating extension ${ext.packageJSON.name}:`, err);
        });
      }
    });

    vscode.workspace.getConfiguration('extensions').update('disableExtensions', true, vscode.ConfigurationTarget.Global).then(() => {
      console.log('Disable extensions configuration set.');
    }).catch(err => {
      console.error('Error setting disable extensions configuration:', err);
    });
  });

  context.subscriptions.push(disposable);

  const disposable2 = vscode.commands.registerCommand('performance-optimizer.enableExtensions', () => {
    vscode.extensions.get()!.forEach(ext => {
      if (ext.isActive) {
        ext.deactivate().then(() => {
          console.log(`Extension ${ext.packageJSON.name} is now deactivated.`);
        }).catch(err => {
          console.error(`Error deactivating extension ${ext.packageJSON.name}:`, err);
        });
      }
    });

    vscode.workspace.getConfiguration('extensions').update('disableExtensions', false, vscode.ConfigurationTarget.Global).then(() => {
      console.log('Enable extensions configuration set.');
    }).catch(err => {
      console.error('Error setting enable extensions configuration:', err);
    });
  });

  context.subscriptions.push(disposable2);
}

export function deactivate() {}
```

### Explanation

1. **package.json**:
   - `name`: The name of the extension.
   - `version`: The version of the extension.
   - `publisher`: Your publisher information.
   - `main`: The main file of the extension, which is `extension.js`.
   - `scripts`:
     - `start`: Starts VS Code with debugging enabled.
     - `pack`: Compiles TypeScript to JavaScript and bundles it using Rollup.

2. **main code (extension.js)**:
   - Activates all extensions in your workspace when the command `performance-optimizer.disableExtensions` is called.
   - Deactivates all extensions when the command `performance-optimizer.enableExtensions` is called.
   - It sets a global configuration to enable or disable extensions, which can be controlled by the user through VS Code settings.

You can further expand this example by adding more commands to optimize specific features of your VS Code environment.
