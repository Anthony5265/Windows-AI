### Main Code (vscode-google-chrome-extension.ts)

```typescript
import * as vscode from 'vscode';

export function activate(context: vscode.ExtensionContext) {
  console.log('Extension activated');

  context.subscriptions.push(vscode.commands.registerCommand(
    'generate-chrome-extension',
    async () => {
      const extensionName = await vscode.window.showInputBox({
        prompt: 'Enter the name of your Chrome Extension:',
        placeHolderText: 'My Extension'
      });

      if (!extensionName) {
        return;
      }

      const manifestContent = `
{
  "manifest_version": 3,
  "name": "${extensionName}",
  "version": "1.0",
  "description": "A simple Chrome extension.",
  "permissions": ["activeTab", "<all_urls>"],
  "icons": {
    "48": "images/icon-48.png",
    "128": "images/icon-128.png"
  }
}
`;

      const manifestPath = vscode.workspace.workspaceFolders?.[0].uri.fsPath + '/manifest.json';
      await vscode.workspace.fs.writeFile(manifestPath, manifestContent);

      const packageJsonContent = {
        name: extensionName,
        version: '1.0',
        description: "A simple Chrome extension.",
        author: "Your Name",
        license: "MIT"
      };

      const packagePath = vscode.workspace.workspaceFolders?.[0].uri.fsPath + '/package.json';
      await vscode.workspace.fs.writeFile(packagePath, JSON.stringify(packageJsonContent));

      vscode.window.showInformationMessage(`Extension generated at ${extensionName}`);
    }
  ));
}
```

### Explanation

1. **Package.json**:
   - Defines the extension's metadata such as name, version, author, and dependencies.
   - Specifies activation events to run the extension when it is activated.
   - Links the main script file.

2. **main.ts**:
   - Registers a command `generate-chrome-extension` that prompts the user for an extension name.
   - Creates a new `manifest.json` file with basic settings and saves it in the workspace folder.
   - Creates a `package.json` file with metadata about the extension and saves it in the workspace folder.
   - Shows a success message to the user.

3. Build the extension using `tsc`.
4. Run the extension using `node out/src/vscodeGoogleChromeExtension.js`.
