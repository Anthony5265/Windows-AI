```javascript
const vscode = require('vscode');
const request = require('request');

class RequestResponseCacheExtension {

  constructor(context) {
    this.context = context;
  }

  activate() {
    const disposable = vscode.workspace.registerTextDocumentContentProvider({
      scheme: 'https',
      provideTextDocumentContent(document, token) {
        const uri = document.uri;
        if (this.cache.has(uri)) {
          return Promise.resolve(this.cache.get(uri));
        }
        return fetch(uri.toString())
          .then(response => response.text())
          .then(text => {
            this.cache.set(uri, text);
            return text;
          })
          .catch(error => {
            vscode.window.showErrorMessage('Failed to fetch data: ' + error.message);
            throw error;
          });
      }
    });

    context.subscriptions.push(disposable);
  }

  deactivate() {
    // Cleanup code if needed
  }
}

module.exports = RequestResponseCacheExtension;
```

This VS Code extension provides a text document content provider for the `https` protocol, caching responses. It fetches remote data using the `request` library and stores it in memory (using an object), which allows subsequent requests to return cached data without making additional network calls.
