```javascript
const vscode = require('vscode');
const { extractLinks } = require('./extract-links');

exports.activate = function() {
  let disposable = vscode.commands.registerCommand('link-extractor.extractLinks', async () => {
    const document = vscode.window.activeTextEditor.document;
    const links = await extractLinks(document.getText());
    vscode.workspace.openTextDocument({ uri: vscode.Uri.parse(`file://${document.fileName}`), content: links.join('\n') }).then(doc => {
      vscode.window.showTextDocument(doc);
    });
  });

  this.subscriptions.push(disposable);
};

exports.deactivate = function() {
  // Clean up anything that was registered
};
```
