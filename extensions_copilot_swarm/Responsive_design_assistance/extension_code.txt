import * as vscode from 'vscode';

export function activate(context: vscode.ExtensionContext) {
  const disposable = vscode.commands.registerCommand('responsiveDesignAssistance.checkResponsive', () => {
    vscode.window.showInformationMessage('Checking responsive design...');
    // Add your logic here to check responsiveness
    setTimeout(() => {
      vscode.window.showInformationMessage('Responsive design checks complete!');
    }, 1000);
  });

  context.subscriptions.push(disposable);
}

export function deactivate() { }
