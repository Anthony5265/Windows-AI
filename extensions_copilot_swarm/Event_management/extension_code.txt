And here is the `main` code (`src/extension.js`):

```javascript
import * as vscode from 'vscode';

export function activate(context: vscode.ExtensionContext) {
  // Register a command that allows users to print the event manager in the console
  context.subscriptions.push(
    vscode.commands.registerCommand('event-manager.printEvent', () => {
      const msg = "Hello, Event Manager! This is a test message from the VS Code extension.";
      vscode.window.showInformationMessage(msg);
    })
  );

  // Register an event listener to track file changes
  let disposable = vscode.workspace.onDidChangeFiles(event => {
    if (event.auditFlags) {
      console.log('File changed:', event.files.map(file => file.name));
    }
  });

  context.subscriptions.push(disposable);

  console.log('Event Manager extension activated');
}

export function deactivate() {
  console.log('Event Manager extension deactivated');
}
```

1. A command that prints a message to the VS Code status bar.
2. An event listener that logs file changes in the console when files are modified.
