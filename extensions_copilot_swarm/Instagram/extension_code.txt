### main.js

```javascript
const vscode = require('vscode');
const axios = require('axios');
const cheerio = require('cheerio');
const pathToRegexp = require('path-to-regexp');
const _ = require('lodash');

class InstagramExtension {
  constructor() {
    this._command = vscode.commands.registerCommand('instagram.openInstagram', async () => {
      await vscode.env.openExternal(vscode.Uri.parse('https://www.instagram.com'));
    });

    this._extensionContext.subscriptions.push(this._command);
  }

  activate(context) {
    console.log('Extension activated!');
  }
}

module.exports = InstagramExtension;
```
